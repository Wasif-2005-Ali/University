var randomNumber, guess, attempts;

function mathRandomInt(a, b) {
  if (a > b) {
    // Swap a and b to ensure a is smaller.
    var c = a;
    a = b;
    b = c;
  }
  return Math.floor(Math.random() * (b - a + 1) + a);
}


randomNumber = mathRandomInt(1, 10);
guess = 0;
attempts = 0;
while (!(guess == randomNumber)) {
  guess = window.prompt('Enter your guess (1–10):');
  attempts = (typeof attempts === 'number' ? attempts : 0) + 1;
  if (guess == randomNumber) {
    window.alert('You guessed it right!');
    window.alert('<audio src=\'https://actions.google.com/sounds/v1/cartoon/cartoon_boing.ogg\' autoplay></audio>');
    window.alert('✨✨✨ Correct Guess! ✨✨✨');
  } else {
    if (guess > randomNumber) {
      window.alert('Try a smaller number!');
    } else {
      window.alert('Try a bigger number!');
    }
  }
}
window.alert(['You got it in  ',attempts,'  tries!!'].join(''));
